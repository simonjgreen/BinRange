cmake_minimum_required(VERSION 3.20.0)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(binrange_k4w_tag)
zephyr_get(BINRANGE_RADIO_DIAGNOSTIC SYSBUILD LOCAL)
option(BINRANGE_RADIO_DIAGNOSTIC "Read radio identity only, no transmissions" ON)
if(BINRANGE_RADIO_DIAGNOSTIC)
  target_sources(app PRIVATE src/main.c)
else()
  target_sources(app PRIVATE src/ranging_main.cpp src/radio_port.cpp src/protocol.c
    ../shared/Dw3000/src/dw3000_device_api.cpp)
  target_include_directories(app PRIVATE ../shared/Dw3000/src)
  target_compile_definitions(app PRIVATE BINRANGE_ZEPHYR=1)
endif()
if(CONFIG_MCUMGR)
  target_sources(app PRIVATE src/ota_runtime.c src/update_policy.c src/motion_policy.c src/tag_power.c)
  target_compile_definitions(app PRIVATE BINRANGE_OTA=1)
endif()
